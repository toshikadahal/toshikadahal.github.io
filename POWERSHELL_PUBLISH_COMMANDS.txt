# Fresh publish script for Dr. Anuj Devkota website
# Save the ZIP in Downloads, extract it, then run these commands from the extracted folder.

git init
git branch -M main
git add .
git commit -m "Launch modern professional ENT website"
gh repo create anujdevkota.github.io --public --description "Professional ENT website of Dr. Anuj Devkota" --source . --remote origin --push

$PagesBody = '{"source":{"branch":"main","path":"/"}}'
$PagesBody | gh api --method POST "repos/anujdevkota/anujdevkota.github.io/pages" --input -
gh repo edit anujdevkota/anujdevkota.github.io --homepage "https://anujdevkota.github.io"
